Use this code snippet in your projects. The JavaScript automatically updates all text blocks based on search parameters.

<!-- Dynamic Text Block Template --> <div class="text-block"> <h1>Find the Best <span class="dynamic-value" data-field="service">plumber</span> in <span class="dynamic-value" data-field="location">Upstate New York</span></h1> </div> <script> // Configuration const searchParams = { service: 'plumber', location: 'Upstate New York', urgency: 'next-day', rating: '4' }; // Update function function updateTextBlocks(params) { // Service variations const service = params.service.toLowerCase(); const servicePlural = pluralize(service); const serviceTitle = capitalize(service); const servicePluralTitle = capitalize(servicePlural); // Urgency text const urgencyMap = { 'emergency': 'emergency', 'same-day': 'same day', 'next-day': 'next day', 'scheduled': 'scheduled' }; const urgencyText = urgencyMap[params.urgency] || params.urgency; const urgencyTextTitle = capitalize(urgencyText); // Rating text const ratingText = params.rating === 'any' ? 'all' : params.rating + '+'; // Update all dynamic values const fields = { 'service': service, 'service-plural': servicePlural, 'service-title': serviceTitle, 'service-plural-title': servicePluralTitle, 'location': params.location, 'urgency-text': urgencyText, 'urgency-text-title': urgencyTextTitle, 'rating': ratingText }; Object.keys(fields).forEach(field => { document.querySelectorAll(`[data-field="${field}"]`).forEach(el => { el.textContent = fields[field]; }); }); } // Helper functions function pluralize(word) { const exceptions = { 'plumber': 'plumbers', 'electrician': 'electricians', 'contractor': 'contractors' }; return exceptions[word] || word + 's'; } function capitalize(str) { return str.charAt(0).toUpperCase() + str.slice(1); } // Call on page load updateTextBlocks(searchParams); </script>
